FRA_debug_effect3 = {
	every_possible_country = {
		limit = { has_character = FRA_charles_de_gaulle }
		FRA_charles_de_gaulle = {
			set_nationality = ROOT
		}
	}
}
FRA_change_rol_anger = {
	if = {
		limit = {
			has_idea = FRA_resistance_of_locals
			OR = {
				check_variable = { FRA_anger_change > 0 }
				AND = {
					check_variable = { FRA_anger_change < 0 }
					check_variable = { FRA_rol_anger > 0 }
				}
			}
		}

		set_temp_variable = { FRA_anger_lower_limit = 1 }
		subtract_from_temp_variable = { FRA_anger_lower_limit = FRA_rol_anger }

		set_temp_variable = { FRA_anger_upper_limit = 8 }
		subtract_from_temp_variable = { FRA_anger_upper_limit = FRA_rol_anger }

		# avoid going < 1 or > 10
		clamp_temp_variable = {
			var = FRA_anger_change
			min = FRA_anger_lower_limit
			max = FRA_anger_upper_limit
		}

		if = {
			limit = {
				OR = {
					NOT = { has_variable = FRA_rol_anger } #bypass the check when initialising
					NOT = { check_variable = { FRA_anger_change = 0 } }
				}
			}
			if = {
				limit = { check_variable = { FRA_anger_change > 0 } }
				custom_effect_tooltip = FRA_resistance_of_locals_up
				log = "Popular Anger Increased"
			}
			else = {
				custom_effect_tooltip = FRA_resistance_of_locals_down
				log = "Popular Anger Decreased"
			}

			# for the tooltips
			set_temp_variable = { FRA_rol_anger_pp_temp = 0.01 }
			multiply_temp_variable = { FRA_rol_anger_pp_temp = FRA_anger_change }

			set_temp_variable = { FRA_rol_anger_stability_temp = 0.03 }
			multiply_temp_variable = { FRA_rol_anger_stability_temp = FRA_anger_change }

			set_temp_variable = { FRA_rol_anger_factory_output_temp = -0.03 }
			multiply_temp_variable = { FRA_rol_anger_factory_output_temp = FRA_anger_change }

			# set the new variables
			add_to_variable = { FRA_rol_anger = FRA_anger_change }
			add_to_variable = { FRA_rol_pp = FRA_rol_anger_pp_temp }
			add_to_variable = { FRA_rol_stability = FRA_rol_anger_factory_output_temp }
			add_to_variable = { FRA_rol_resist = FRA_rol_anger_stability_temp }
			add_to_variable = { FRA_rol_cg = FRA_rol_anger_factory_output_temp }

			log = "Current anger levels: Military Anger = [?RUS_military_anger], Popular Anger = [?SHA_aup_anger], Right-Anger = [?RUS_right_anger], Left-Anger = [?RUS_left_anger]"
		}
	}
}
