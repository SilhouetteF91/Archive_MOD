GEH_ako_route_dating_sim_ending_effect = {
	set_global_flag = GEH_ako_route_dating_sim_ending_flag
	if = {
		limit = {
			OR = {
				is_power_balance_in_range = {
					id = GEH_ako_love_balance
					range = GEH_hina_love_ako_side_low
				}
				is_power_balance_in_range = {
					id = GEH_ako_love_balance
					range = GEH_hina_love_ako_side_win
				}
			}
		}
		country_event = { id = GEH.event.11 }
	}
	else_if = {
		limit = {
			OR = {
				is_power_balance_in_range = {
					id = GEH_ako_love_balance
					range = GEH_hina_hate_ako_side_low
				}
				is_power_balance_in_range = {
					id = GEH_ako_love_balance
					range = GEH_hina_hate_ako_side_win
				}
			}
		}
		country_event = { id = GEH.event.9 }
	}
	else_if = {
		limit = {
			is_power_balance_in_range = {
				id = GEH_ako_love_balance
				range = GEH_ako_love_netural_side
			}
		}
		country_event = { id = GEH.event.8 }
	}

	remove_power_balance = {
		id = GEH_ako_love_balance
	}
}

GEH_OGAS_increase_effect = {
	if = {
		limit = { 
			NOT = { has_idea = GEH_ogas_1 } 
			NOT = { has_idea = GEH_ogas_2 } 
			NOT = { has_idea = GEH_ogas_3 } 
			NOT = { has_idea = GEH_ogas_4 } 
		}
		add_ideas = GEH_ogas_1
	}
	else_if = {
		limit = { has_idea = GEH_ogas_1 }
		swap_ideas = {
			add_idea = GEH_ogas_2
			remove_idea = GEH_ogas_1
		}
	}
	else_if = {
		limit = { has_idea = GEH_ogas_2 }
		swap_ideas = {
			add_idea = GEH_ogas_3
			remove_idea = GEH_ogas_2
		}
	}
	else_if = {
		limit = { has_idea = GEH_ogas_3 }
		swap_ideas = {
			add_idea = GEH_ogas_4
			remove_idea = GEH_ogas_3
		}
	}
	else = {
		effect_tooltip = {
			add_ideas = GEH_ogas_4
		}
	}
}
