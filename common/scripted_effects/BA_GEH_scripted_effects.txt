GEH_OGAS_increase_effect = {
	if = {
		limit = { 
			NOT = { has_idea = GEH_ogas_1 } 
			NOT = { has_idea = GEH_ogas_2 } 
			NOT = { has_idea = GEH_ogas_3 } 
			NOT = { has_idea = GEH_ogas_4 } 
		}
		add_ideas = GEH_ogas_1
	}
	else_if = {
		limit = { has_idea = GEH_ogas_1 }
		swap_ideas = {
			add_idea = GEH_ogas_2
			remove_idea = GEH_ogas_1
		}
	}
	else_if = {
		limit = { has_idea = GEH_ogas_2 }
		swap_ideas = {
			add_idea = GEH_ogas_3
			remove_idea = GEH_ogas_2
		}
	}
	else_if = {
		limit = { has_idea = GEH_ogas_3 }
		swap_ideas = {
			add_idea = GEH_ogas_4
			remove_idea = GEH_ogas_3
		}
	}
	else = {
		effect_tooltip = {
			add_ideas = GEH_ogas_4
		}
	}
}

    if = {
        limit = {
            has_idea = GEH_military_tradition_1
        }
        swap_ideas = {
            remove_idea = GEH_military_tradition_1
            add_idea = GEH_military_tradition_2
        }
    }
	else_if = {
		limit = {
			has_idea = GEH_military_tradition_2
		}
		swap_ideas = {
			remove_idea = GEH_military_tradition_2
			add_idea = GEH_military_tradition_3
		}
	}
	else = {
		add_breakthrough_progress = {
			specialization = specialization_land
			value = 0.5
		}
	}
}
