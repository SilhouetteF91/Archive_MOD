# GEM의 기본 유닛 생산: 보병 중심
GEM_unit_production_early = {
	allowed = { original_tag = GEM }
	enable = {
		date < 1940.1.1 # 1940년 이전까지 적용
	}
	abort_when_not_enabled = yes

	# --- 육군: 보병에만 집중 ---
	ai_strategy = {
		type = role_ratio
		id = infantry # 보병 비율을 압도적으로 높게 설정
		value = 100
	}
	ai_strategy = {
		type = role_ratio
		id = armor # 기갑 생산 억제
		value = -100
	}
    ai_strategy = {
		type = role_ratio
		id = mechanized # 기계화보병 생산 억제
		value = -100
	}

    ai_strategy = {
        type = role_ratio
        id = garrison
        # 음수 값을 주어 AI가 원하는 주둔군/해안방어 사단 수를 줄입니다.
        # 예: -30은 평소보다 30% 적은 수의 사단을 원하게 됨을 의미합니다.
        value = -60
    }
	
	# --- 공군: 방어적인 전투기만 생산 ---
	ai_strategy = {
		type = unit_ratio
		id = fighter # 제공권 방어를 위한 전투기만 생산
		value = 100
	}
	ai_strategy = {
		type = unit_ratio
		id = cas # CAS, 폭격기 생산은 0으로 설정
		value = 0
	}
	ai_strategy = {
		type = unit_ratio
		id = tactical_bomber
		value = 0
	}

	# --- 장비 생산: 보병과 포병 장비 우선 ---
	ai_strategy = {
		type = equipment_production_factor
		id = infantry
		value = 50
	}
	ai_strategy = {
		type = equipment_production_factor
		id = artillery
		value = 25
	}
    ai_strategy = {
		type = equipment_production_factor
		id = armor # 기갑 장비 생산 억제
		value = -50
	}
}

# 초기 방어 전략: 내륙에서 버티기
GEM_early_defense = {
    allowed = { original_tag = GEM }
    enable = { date < 1940.6.1 } # 1940년 6월까지 적용
    abort_when_not_enabled = yes

    # 작전 계획을 '신중한(careful)'으로 설정하여 소모를 최소화
    ai_strategy = {
		type = front_control
		ratio = 0.5
		priority = 100
		ordertype = front
		execution_type = careful
	}

    # 내륙에 방어 병력을 미리 배치하여 방어선 붕괴 방지
    ai_strategy = {
		type = put_unit_buffers
		states = { # (가상의 내륙 주 ID를 기입)
			55
			59
			51
		}
		ratio = 0.20 # 군대의 20%를 예비대로 배치
		area = asia
	}
}
# 1940년 이후 군비 증강 계획: 기갑

		date > 1940.1.1
	}
	abort = { always = no }

		type = role_ratio
		id = armor

	}
    ai_strategy = {
		type = role_ratio
		id = mechanized
		value = 10 # 기계화보병도 함께 생산
	}

		type = equipment_production_factor
		id = armor
		value = 40
	}

    # 중형 전차 생산라인 최소 5개 유지
    ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = medium_tank_chassis
		value = 5
	}
}
