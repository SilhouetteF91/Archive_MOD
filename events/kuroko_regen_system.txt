add_namespace = kuroko_regen

# 쿠로코 존재 확인 이벤트 (매 30일)
country_event = {
	id = kuroko_regen.1
	title = kuroko_regen.1.t
	desc = kuroko_regen.1.d
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes
	hidden = yes

	trigger = {
		tag = GEI
		has_tech = kuroko_raid_technology
		NOT = { has_country_flag = kuroko_regen_in_progress }
	}

	immediate = {
		# 쿠로코 사단 존재 여부 확인 (쿠로코 캡이 1 이상이면 존재)
		if = {
			limit = {
				check_variable = { kuroko_infantry_cap > 0 }
				has_template = "Shiroko Terror"
			}
			# 쿠로코가 존재함 - 다음 확인 스케줄
			country_event = { id = kuroko_regen.1 days = 30 }
		}
		else = {
			# 쿠로코가 없음 - 리젠 타이머 시작
			country_event = { id = kuroko_regen.2 hours = 1 }
		}
	}
}

# 쿠로코 리젠 실행 (180일 후)
country_event = {
	id = kuroko_regen.3
	title = kuroko_regen.3.t
	desc = kuroko_regen.3.d
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes

	trigger = {
		tag = GEI
		has_tech = kuroko_raid_technology
		has_country_flag = kuroko_regen_in_progress
	}

	option = {
		name = kuroko_regen.3.a	# "죽음의 여신이 돌아왔다!"

		clr_country_flag = kuroko_regen_in_progress
		
		# 쿠로코 사단 리젠 실행
		hidden_effect = {
			# 색채 영토에서 적절한 주 찾기
			random_owned_controlled_state = {
				limit = {
					is_core_of = GEI
					is_coastal = yes
				}
				prioritize = { 100 }  # 색채 본토 우선
				save_event_target_as = kuroko_spawn_location
			}
			
			# 해안 지역이 없으면 아무 핵심 영토나
			if = {
				limit = {
					NOT = { has_event_target = kuroko_spawn_location }
				}
				random_owned_controlled_state = {
					limit = {
						is_core_of = GEI
					}
					save_event_target_as = kuroko_spawn_location
				}
			}
		}
		
		# 쿠로코 특수 사단 1개 추가
		custom_effect_tooltip = kivotos_kuroko_add1_tt
		set_variable = { BA_kuroko_infantry_cap_edit_var = 1 }
		BA_kuroko_infantry_cap_edit_effect = yes
		
		# 배치 이벤트 실행
		country_event = { id = kuroko_regen.4 hours = 1 }
		
		# 다음 확인 스케줄 재시작
		country_event = { id = kuroko_regen.1 days = 30 }
		
		# 리젠 성공 보너스
		add_command_power = 50
		add_war_support = 0.05
	}
}

		# 배치 성공 보너스
		add_command_power = 75
		add_war_support = 0.05
	}
}

# 긴급 쿠로코 소환 (색채 본토가 위험할 때)
country_event = {
	id = kuroko_regen.5
	title = kuroko_regen.5.t
	desc = kuroko_regen.5.d
	picture = GFX_report_event_generic_read_write

	trigger = {
		tag = GEI
		has_tech = kuroko_raid_technology
		NOT = { has_country_flag = kuroko_regen_in_progress }
		NOT = {
			check_variable = { kuroko_infantry_cap > 0 }
			has_template = "Shiroko Terror"
		}
		OR = {
			# 색채가 전쟁 중이고 전세가 불리할 때
			AND = {
				has_war = yes
				surrender_progress > 0.15
			}
			# 색채 본토가 적에게 점령당했을 때
			NOT = { controls_state = 100 }
		}
	}

	mean_time_to_happen = {
		days = 30
		modifier = {
			factor = 0.5
			surrender_progress > 0.25
		}
		modifier = {
			factor = 0.3
			surrender_progress > 0.5
		}
	}

	option = {
		name = kuroko_regen.5.a	# "위기의 순간, 죽음의 여신을 소환한다!"

		# 즉시 쿠로코 리젠
		custom_effect_tooltip = kivotos_kuroko_add1_tt
		set_variable = { BA_kuroko_infantry_cap_edit_var = 1 }
		BA_kuroko_infantry_cap_edit_effect = yes
		
		# 긴급 배치
		random_owned_controlled_state = {
			limit = {
				is_core_of = GEI
			}
			create_unit = {
				division = "division_template = \"Shiroko Terror\" start_experience_factor = 1.0"
				owner = GEI
				count = 1
			}
		}
		
		# 긴급 소환 보너스
		add_command_power = 100
		add_war_support = 0.10
		# 긴급 소환 임시 보너스
		add_command_power = 50
		
		# 다음 확인 스케줄
		country_event = { id = kuroko_regen.1 days = 30 }
	}

	option = {
		name = kuroko_regen.5.b	# "아직 때가 아니다"
		
		# 일반 리젠 타이머 시작
		country_event = { id = kuroko_regen.2 hours = 1 }
	}
}
